cmake_minimum_required(VERSION 3.22.1)
project(harumi_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_COMMON LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_BUILD_UI LLAMA_OPENSSL GGML_NATIVE GGML_OPENMP GGML_LLAMAFILE GGML_CPU_KLEIDIAI GGML_VULKAN GGML_CUDA GGML_METAL)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU ON CACHE BOOL "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp llama-runtime)
add_library(harumi SHARED harumi.cpp)
target_link_libraries(harumi PRIVATE llama android log)
target_compile_options(harumi PRIVATE -O3)
